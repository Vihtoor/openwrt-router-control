   308	@OptIn(ExperimentalMaterial3Api::class)
   309	@Composable
   310	fun ImportKeyDialog(
   311	    onDismiss: () -> Unit,
   312	    onKeyImported: (Long) -> Unit
   313	) {
   314	    val context = LocalContext.current
   315	    val coroutineScope = rememberCoroutineScope()
   316	    
   317	    var inputMode by remember { mutableStateOf(1) } // 0 = file, 1 = text
   318	    var privateKeyText by remember { mutableStateOf("") }
   319	    var privateKeyBytes by remember { mutableStateOf<ByteArray?>(null) }
   320	    var fileName by remember { mutableStateOf<String?>(null) }
   321	    
   322	    var alias by remember { mutableStateOf("") }
   323	    var passphrase by remember { mutableStateOf("") }
   324	    var publicKeyText by remember { mutableStateOf("") }
   325	    
   326	    var parsedKeyType by remember { mutableStateOf<Int?>(null) }
   327	    var isEncrypted by remember { mutableStateOf(false) }
   328	    var isDecrypted by remember { mutableStateOf(true) }
   329	    
   330	    var errorMsg by remember { mutableStateOf<String?>(null) }
   331	    
   332	    val launcher = androidx.activity.compose.rememberLauncherForActivityResult(androidx.activity.result.contract.ActivityResultContracts.GetContent()) { uri: android.net.Uri? ->
   333	        if (uri != null) {
   334	            try {
   335	                val inputStream = context.contentResolver.openInputStream(uri)
   336	                val bytes = inputStream?.readBytes()
   337	                inputStream?.close()
   338	                if (bytes != null) {
   339	                    privateKeyBytes = bytes
   340	                    fileName = uri.path?.substringAfterLast('/') ?: "file"
   341	                    inputMode = 0
   342	                }
   343	            } catch (e: Exception) {
   344	                errorMsg = "Ошибка чтения файла: ${e.message}"
   345	            }
   346	        }
   347	    }
   348	    
   349	    LaunchedEffect(privateKeyText, privateKeyBytes, inputMode) {
   350	        val bytes = if (inputMode == 1) {
   351	            if (privateKeyText.isNotBlank()) privateKeyText.toByteArray(Charsets.UTF_8) else null
   352	        } else {
   353	            privateKeyBytes
   354	        }
   355	        
   356	        if (bytes == null || bytes.isEmpty()) {
   357	            parsedKeyType = null
   358	            isEncrypted = false
   359	            isDecrypted = true
   360	            errorMsg = null
   361	            return@LaunchedEffect
   362	        }
   363	        
   364	        try {
   365	            val jsch = JSch()
   366	            val kp = KeyPair.load(jsch, bytes, null)
   367	            parsedKeyType = kp.keyType
   368	            isEncrypted = kp.isEncrypted
   369	            isDecrypted = !kp.isEncrypted
   370	            errorMsg = null
   371	            
   372	            if (!kp.isEncrypted) {
   373	                val pubOut = java.io.ByteArrayOutputStream()
   374	                kp.writePublicKey(pubOut, "imported-key")
   375	                val pubStr = String(pubOut.toByteArray())
   376	                if (pubStr.isNotBlank()) {
   377	                    publicKeyText = pubStr
   378	                }
   379	            }
   380	            kp.dispose()
   381	        } catch (e: Exception) {
   382	            parsedKeyType = null
   383	            isEncrypted = false
   384	            isDecrypted = true
   385	            errorMsg = "Не удалось распознать формат ключа"
   386	        }
   387	    }
   388	    
   389	    LaunchedEffect(passphrase, parsedKeyType, isEncrypted) {
   390	        if (isEncrypted && passphrase.isNotEmpty()) {
   391	            val bytes = if (inputMode == 1) privateKeyText.toByteArray(Charsets.UTF_8) else privateKeyBytes
   392	            if (bytes != null) {
   393	                try {
   394	                    val jsch = JSch()
   395	                    val kp = KeyPair.load(jsch, bytes, null)
   396	                    if (kp.decrypt(passphrase)) {
   397	                        isDecrypted = true
   398	                        errorMsg = null
   399	                        val pubOut = java.io.ByteArrayOutputStream()
   400	                        kp.writePublicKey(pubOut, "imported-key")
   401	                        val pubStr = String(pubOut.toByteArray())
   402	                        if (pubStr.isNotBlank() && publicKeyText.isBlank()) {
   403	                            publicKeyText = pubStr
   404	                        }
   405	                    } else {
   406	                        isDecrypted = false
   407	                    }
   408	                    kp.dispose()
   409	                } catch (e: Exception) {
   410	                    isDecrypted = false
   411	                }
   412	            }
   413	        }
   414	    }
   415	    
   416	    var isSaving by remember { mutableStateOf(false) }
   417	    
   418	    Dialog(
   419	        onDismissRequest = { if (!isSaving) onDismiss() },
   420	        properties = DialogProperties(usePlatformDefaultWidth = false)
   421	    ) {
   422	        Surface(
   423	            modifier = Modifier
   424	                .fillMaxWidth(0.95f)
   425	                .fillMaxHeight(0.95f),
   426	            shape = RoundedCornerShape(16.dp),
   427	            color = MaterialTheme.colorScheme.surface
   428	        ) {
   429	            Column(modifier = Modifier.padding(16.dp)) {
   430	                Text(
   431	                    text = translateText("Импорт ключа", context),
   432	                    style = MaterialTheme.typography.titleLarge,
   433	                    fontWeight = FontWeight.Bold,
   434	                    modifier = Modifier.padding(bottom = 16.dp)
   435	                )
   436	                
   437	                Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(8.dp)) {
   438	                    Button(
   439	                        onClick = { launcher.launch("*/*") },
   440	                        modifier = Modifier.weight(1f).height(40.dp),
   441	                        contentPadding = PaddingValues(0.dp),
   442	                        colors = ButtonDefaults.buttonColors(
   443	                            containerColor = if (inputMode == 0) MaterialTheme.colorScheme.primary.copy(alpha = 0.2f) else MaterialTheme.colorScheme.surfaceVariant,
   444	                            contentColor = if (inputMode == 0) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.onSurface
   445	                        ),
   446	                        shape = RoundedCornerShape(20.dp),
   447	                        border = if (inputMode == 0) androidx.compose.foundation.BorderStroke(1.dp, MaterialTheme.colorScheme.primary) else null
   448	                    ) {
   449	                        Icon(Icons.Default.Upload, contentDescription = null, modifier = Modifier.size(16.dp))
   450	                        Spacer(modifier = Modifier.width(8.dp))
   451	                        Text(translateText("Из файла", context), fontSize = 13.sp)
   452	                    }
   453	                    Button(
   454	                        onClick = { inputMode = 1 },
   455	                        modifier = Modifier.weight(1f).height(40.dp),
   456	                        contentPadding = PaddingValues(0.dp),
   457	                        colors = ButtonDefaults.buttonColors(
   458	                            containerColor = if (inputMode == 1) MaterialTheme.colorScheme.primary.copy(alpha = 0.2f) else MaterialTheme.colorScheme.surfaceVariant,
   459	                            contentColor = if (inputMode == 1) MaterialTheme.colorScheme.primary else MaterialTheme.colorScheme.onSurface
   460	                        ),
   461	                        shape = RoundedCornerShape(20.dp),
   462	                        border = if (inputMode == 1) androidx.compose.foundation.BorderStroke(1.dp, MaterialTheme.colorScheme.primary) else null
   463	                    ) {
   464	                        Icon(Icons.Default.ContentPaste, contentDescription = null, modifier = Modifier.size(16.dp))
   465	                        Spacer(modifier = Modifier.width(8.dp))
   466	                        Text(translateText("Текстом", context), fontSize = 13.sp)
   467	                    }
   468	                }
   469	                
   470	                Spacer(modifier = Modifier.height(16.dp))
   471	                
   472	                val scrollState = androidx.compose.foundation.rememberScrollState()
   473	                Column(modifier = Modifier.weight(1f).verticalScroll(scrollState)) {
   474	                    
   475	                    if (inputMode == 1) {
   476	                        OutlinedTextField(
   477	                            value = privateKeyText,
   478	                            onValueChange = { privateKeyText = it },
   479	                            label = { Text(translateText("Приватный ключ", context)) },
   480	                            modifier = Modifier.fillMaxWidth().heightIn(min = 100.dp, max = 200.dp),
   481	                            textStyle = androidx.compose.ui.text.TextStyle(fontFamily = androidx.compose.ui.text.font.FontFamily.Monospace, fontSize = 12.sp)
   482	                        )
   483	                        Spacer(modifier = Modifier.height(8.dp))
   484	                    } else if (fileName != null) {
   485	                        Row(
   486	                            modifier = Modifier
   487	                                .fillMaxWidth()
   488	                                .background(MaterialTheme.colorScheme.surfaceVariant, RoundedCornerShape(8.dp))
   489	                                .padding(12.dp),
   490	                            verticalAlignment = Alignment.CenterVertically
   491	                        ) {
   492	                            Icon(Icons.Default.InsertDriveFile, contentDescription = null)
   493	                            Spacer(modifier = Modifier.width(12.dp))
   494	                            Text(fileName!!, style = MaterialTheme.typography.bodyMedium)
   495	                        }
   496	                        Spacer(modifier = Modifier.height(8.dp))
   497	                    }
   498	                    
   499	                    if (parsedKeyType != null) {
   500	                        val typeName = when(parsedKeyType) {
   501	                            KeyPair.RSA -> "RSA"
   502	                            KeyPair.DSA -> "DSA"
   503	                            KeyPair.ECDSA -> "ECDSA"
   504	                            KeyPair.ED25519 -> "ED25519"
   505	                            else -> "UNKNOWN"
   506	                        }
   507	                        Row(
   508	                            modifier = Modifier
   509	                                .fillMaxWidth()
   510	                                .background(androidx.compose.ui.graphics.Color(0xFF1C2E23), RoundedCornerShape(8.dp))
   511	                                .padding(12.dp),
   512	                            verticalAlignment = Alignment.CenterVertically
   513	                        ) {
   514	                            Icon(Icons.Default.CheckCircle, contentDescription = null, tint = androidx.compose.ui.graphics.Color(0xFF6FD394))
   515	                            Spacer(modifier = Modifier.width(12.dp))
   516	                            Column {
